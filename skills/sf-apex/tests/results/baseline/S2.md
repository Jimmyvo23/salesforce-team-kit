**`force-app/main/default/classes/OrderErpService.cls`**
```apex
public with sharing class OrderErpService {
    @TestVisible
    private static final String ENDPOINT = 'callout:ERP/api/orders';
    private static final Integer TIMEOUT_MS = 20000;

    public class ErpException extends Exception {}

    // Call from the trigger; callouts are not allowed in trigger context, so this enqueues the work.
    public static void enqueueOrders(Set<Id> orderIds) {
        if (orderIds == null || orderIds.isEmpty() || System.isFuture() || System.isQueueable()) {
            return;
        }
        System.enqueueJob(new PostOrdersJob(orderIds));
    }

    public static void postOrders(Set<Id> orderIds) {
        List<Order> orders = [
            SELECT Id, OrderNumber, AccountId, Status, Type, EffectiveDate, TotalAmount, CurrencyIsoCode
            FROM Order
            WHERE Id IN :orderIds
            WITH USER_MODE
        ];
        for (Order o : orders) {
            postOrder(o);
        }
    }

    public static void postOrder(Order o) {
        HttpRequest req = new HttpRequest();
        req.setEndpoint(ENDPOINT);
        req.setMethod('POST');
        req.setHeader('Content-Type', 'application/json');
        req.setTimeout(TIMEOUT_MS);
        req.setBody(toJson(o));

        HttpResponse res = new Http().send(req);
        if (res.getStatusCode() < 200 || res.getStatusCode() >= 300) {
            throw new ErpException(
                'ERP rejected Order ' + o.Id + ': ' + res.getStatusCode() + ' ' + res.getStatus()
            );
        }
    }

    @TestVisible
    private static String toJson(Order o) {
        Map<String, Object> payload = new Map<String, Object>{
            'salesforceId' => o.Id,
            'orderNumber' => o.OrderNumber,
            'accountId' => o.AccountId,
            'status' => o.Status,
            'type' => o.Type,
            'effectiveDate' => o.EffectiveDate,
            'totalAmount' => o.TotalAmount,
            'currencyIsoCode' => o.CurrencyIsoCode
        };
        return JSON.serialize(payload);
    }

    public class PostOrdersJob implements Queueable, Database.AllowsCallouts {
        private final Set<Id> orderIds;

        public PostOrdersJob(Set<Id> orderIds) {
            this.orderIds = orderIds;
        }

        public void execute(QueueableContext ctx) {
            postOrders(orderIds);
        }
    }
}
```

@IsTest
private class OrderErpServiceTest {
    private class MockErp implements HttpCalloutMock {
        Integer status;
        String lastBody;
        MockErp(Integer status) { this.status = status; }
        public HttpResponse respond(HttpRequest req) {
            lastBody = req.getBody();
            System.assertEquals('POST', req.getMethod());
            System.assertEquals(OrderErpService.ENDPOINT, req.getEndpoint());
            HttpResponse res = new HttpResponse();
            res.setStatusCode(status);
            res.setBody('{}');
            return res;
        }
    }

    private static Order makeOrder() {
        Account a = new Account(Name = 'Test Account');
        insert a;
        Order o = new Order(AccountId = a.Id, Status = 'Draft', EffectiveDate = Date.today());
        insert o;
        return o;
    }

    @IsTest
    static void triggerEnqueuesAndPostsJson() {
        MockErp mock = new MockErp(201);
        Test.setMock(HttpCalloutMock.class, mock);
        Test.startTest();
        Order o = makeOrder();
        Test.stopTest();

        System.assertNotEquals(null, mock.lastBody, 'ERP should have been called');
        Map<String, Object> body = (Map<String, Object>) JSON.deserializeUntyped(mock.lastBody);
        System.assertEquals(o.Id, (Id) body.get('salesforceId'));
    }

    @IsTest
    static void nonSuccessResponseThrows() {
        Test.setMock(HttpCalloutMock.class, new MockErp(500));
        Order o = makeOrder();
        Test.startTest();
        try {
            OrderErpService.postOrders(new Set<Id>{ o.Id });
            System.assert(false, 'Expected ErpException');
        } catch (OrderErpService.ErpException e) {
            System.assert(e.getMessage().contains('500'));
        }
        Test.stopTest();
    }
}

Each of the three files also needs a `-meta.xml` file, copied from `OpportunityService.cls-meta.xml`. For the trigger, use `<ApexTrigger>` as the root element instead of `<ApexClass>`. Use API version 62.0 in all three.

**Design notes**
- **Async callout:** Apex doesn't allow callouts directly in trigger context. The trigger enqueues a `Queueable` that implements `Database.AllowsCallouts`, so the Order save doesn't wait on the ERP.
- **Named Credential:** The service calls `callout:ERP/api/orders` instead of hardcoding the URL. You need to create a Named Credential called `ERP` with URL `https://erp.example.com` and its auth settings in the sandbox. This keeps secrets out of code and avoids a Remote Site Setting. If you'd rather hardcode the URL, change `ENDPOINT` and add a Remote Site Setting.
- **Payload fields:** I chose these fields: Id, OrderNumber, AccountId, Status, Type, EffectiveDate, TotalAmount and CurrencyIsoCode. Adjust them to match the ERP's schema. `CurrencyIsoCode` only exists if multi-currency is enabled, so remove it otherwise.
- **Failures:** Non-2xx responses throw `ErpException`, which fails the queueable job and shows up in Apex Jobs. There is no retry or status field on Order yet. If you want that, I'd add an `ERP_Sync_Status__c` field and a `Finalizer`.
- **Trigger scope:** The trigger fires on every insert and update, so each edit re-posts the Order. If the ERP should only receive Orders on certain changes, such as a status change, I can add a filter.
- **Bulk limits:** A single job makes one callout per Order, and Apex allows 100 callouts per transaction. For bulk loads of more than 100 Orders, batch the IDs into chunks.
