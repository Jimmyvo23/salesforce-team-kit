**`triggers/ContactTrigger.trigger`**
```apex
trigger ContactTrigger on Contact (after insert, after update) {
    ContactTriggerHandler.updateAccountLastContactDate(Trigger.new, Trigger.oldMap);
}
```

**`classes/ContactTriggerHandler.cls`**
```apex
public without sharing class ContactTriggerHandler {
    public static void updateAccountLastContactDate(List<Contact> newContacts, Map<Id, Contact> oldMap) {
        Set<Id> accountIds = new Set<Id>();
        for (Contact c : newContacts) {
            if (c.AccountId == null) {
                continue;
            }
            Boolean isInsert = oldMap == null;
            if (isInsert || c.Email != oldMap.get(c.Id).Email) {
                accountIds.add(c.AccountId);
            }
        }
        if (accountIds.isEmpty()) {
            return;
        }

        List<Account> accounts = new List<Account>();
        for (Id accountId : accountIds) {
            accounts.add(new Account(Id = accountId, Last_Contact_Date__c = Date.today()));
        }
        update accounts;
    }
}
```

**`classes/ContactTriggerHandlerTest.cls`**
```apex
@IsTest
private class ContactTriggerHandlerTest {
    private static final Date STALE_DATE = Date.today().addDays(-30);

    @TestSetup
    static void setup() {
        List<Account> accounts = new List<Account>();
        for (Integer i = 0; i < 2; i++) {
            accounts.add(new Account(Name = 'Test Account ' + i, Last_Contact_Date__c = STALE_DATE));
        }
        insert accounts;
    }

    private static Account getAccount(String name) {
        return [SELECT Id, Last_Contact_Date__c FROM Account WHERE Name = :name];
    }

    private static void resetDates() {
        List<Account> accounts = [SELECT Id FROM Account];
        for (Account a : accounts) {
            a.Last_Contact_Date__c = STALE_DATE;
        }
        update accounts;
    }

    @IsTest
    static void insertUpdatesAccountDate() {
        Account acc = getAccount('Test Account 0');

        Test.startTest();
        insert new Contact(LastName = 'Doe', AccountId = acc.Id, Email = 'doe@example.com');
        Test.stopTest();

        System.assertEquals(Date.today(), getAccount('Test Account 0').Last_Contact_Date__c);
        System.assertEquals(STALE_DATE, getAccount('Test Account 1').Last_Contact_Date__c, 'Other accounts untouched');
    }

    @IsTest
    static void emailChangeUpdatesAccountDate() {
        Account acc = getAccount('Test Account 0');
        Contact c = new Contact(LastName = 'Doe', AccountId = acc.Id, Email = 'old@example.com');
        insert c;
        resetDates();

        Test.startTest();
        c.Email = 'new@example.com';
        update c;
        Test.stopTest();

        System.assertEquals(Date.today(), getAccount('Test Account 0').Last_Contact_Date__c);
    }

    @IsTest
    static void nonEmailChangeDoesNotUpdateAccountDate() {
        Account acc = getAccount('Test Account 0');
        Contact c = new Contact(LastName = 'Doe', AccountId = acc.Id, Email = 'doe@example.com');
        insert c;
        resetDates();

        Test.startTest();
        c.FirstName = 'Jane';
        update c;
        Test.stopTest();

        System.assertEquals(STALE_DATE, getAccount('Test Account 0').Last_Contact_Date__c);
    }

    @IsTest
    static void contactWithoutAccountIsIgnored() {
        Test.startTest();
        insert new Contact(LastName = 'Orphan', Email = 'orphan@example.com');
        Test.stopTest();

        System.assertEquals(STALE_DATE, getAccount('Test Account 0').Last_Contact_Date__c);
    }

    @IsTest
    static void bulkInsertAcrossAccounts() {
        List<Account> accounts = [SELECT Id FROM Account];
        List<Contact> contacts = new List<Contact>();
        for (Integer i = 0; i < 200; i++) {
            contacts.add(new Contact(
                LastName = 'Bulk' + i,
                AccountId = accounts[Math.mod(i, accounts.size())].Id,
                Email = 'bulk' + i + '@example.com'
            ));
        }

        Test.startTest();
        insert contacts;
        Test.stopTest();

        for (Account a : [SELECT Last_Contact_Date__c FROM Account]) {
            System.assertEquals(Date.today(), a.Last_Contact_Date__c);
        }
    }
}
```

Each of the three classes and the trigger also needs a `-meta.xml` file with `apiVersion` 62.0 and `status` Active, matching `OpportunityService.cls-meta.xml`. For the trigger, the root element is `ApexTrigger` instead of `ApexClass`.

- **Field metadata:** `Last_Contact_Date__c` isn't in the repo. I assumed it's a Date field. Skip the field file if it already exists in your org, and change the type if it's a DateTime.
- **Sharing:** The handler is `without sharing` so the update still works when the Contact's creator can't edit the Account. Switch it to `with sharing` if you'd rather enforce that.
- **Logic:**
  - It's bulk-safe and does one Account update per transaction, deduplicated by Account.
  - Contacts without an Account are skipped.
  - On update, only an Email change counts. A change from null to a value counts too.
- **Tests:** They cover insert, email change, a non-email change that shouldn't update the date, a Contact with no Account, and a 200-record bulk insert.
