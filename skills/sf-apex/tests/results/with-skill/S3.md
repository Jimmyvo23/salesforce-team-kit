## How it works
- **Batch Apex, not a loop of jobs.** `start` returns a QueryLocator, which handles 3M rows (the limit is 50M). `execute` runs on 200 records at a time.
- **Only changed rows are written.** Unchanged rows are skipped, which cuts nightly DML sharply once the data settles.
- **Partial success.** One bad record doesn't roll back the other 199. Failures are counted and logged in `finish`.
- **The schedulable only starts the batch.** The logic lives in the batch and in `AccountScoring`.
- **System mode, on purpose.** The batch is `without sharing` and uses `AccessLevel.SYSTEM_MODE`, with a comment saying so. That way it recalculates every Account regardless of who scheduled it. If you'd rather it respect the scheduling user's access, switch to `with sharing` and `USER_MODE`. Schedule it as an integration user.

public with sharing class AccountScoring {
    // PLACEHOLDER rule: replace with the real calculation.
    // 1 point per USD 1M of annual revenue (max 70) + 1 point per 10 employees (max 30).
    public static Decimal score(Account a) {
        Decimal revenuePoints = a.AnnualRevenue == null ? 0 : Math.min(70, Math.floor(a.AnnualRevenue / 1000000));
        Decimal employeePoints = a.NumberOfEmployees == null ? 0 : Math.min(30, Math.floor(a.NumberOfEmployees / 10));
        return revenuePoints + employeePoints;
    }

**`classes/AccountScoreBatch.cls`**
```apex
// Org-wide nightly maintenance: runs without sharing and in system mode on purpose, so it
// recalculates every Account regardless of who scheduled it (the field is derived data and
// must be consistent for all users). Schedule it as an integration user.
public without sharing class AccountScoreBatch implements Database.Batchable<SObject>, Database.Stateful {
    private Integer updated = 0;
    private Integer failed = 0;

    public Database.QueryLocator start(Database.BatchableContext bc) {
        return Database.getQueryLocator(
            'SELECT Id, AnnualRevenue, NumberOfEmployees, Score__c FROM Account',
            AccessLevel.SYSTEM_MODE);
    }

    public void execute(Database.BatchableContext bc, List<Account> scope) {
        List<Account> changed = new List<Account>();
        for (Account a : scope) {
            Decimal score = AccountScoring.score(a);
            if (a.Score__c != score) {          // skip unchanged rows to save DML
                a.Score__c = score;
                changed.add(a);
            }
        }
        // Partial success: one bad record must not roll back the other 199.
        List<Database.SaveResult> results = Database.update(changed, false, AccessLevel.SYSTEM_MODE);
        for (Database.SaveResult sr : results) {
            if (sr.isSuccess()) {
                updated++;
            } else {
                failed++;
                System.debug(LoggingLevel.ERROR, 'Score update failed for ' + sr.getId() + ': ' + sr.getErrors()[0].getMessage());
            }
        }
    }

    public void finish(Database.BatchableContext bc) {
        System.debug('AccountScoreBatch finished. Updated: ' + updated + ', failed: ' + failed);
    }
}
```

@IsTest
private class AccountScoreBatchTest {
    @IsTest
    static void batchScores200Accounts() {
        TestDataFactory.accounts(200, true);

        Test.startTest();
        Database.executeBatch(new AccountScoreBatch(), 200);
        Test.stopTest();

        // 5M revenue = 5 points, 100 employees = 10 points
        Assert.areEqual(200, [SELECT COUNT() FROM Account WHERE Score__c = 15], 'All accounts scored 15');
    }

    @IsTest
    static void batchLeavesCorrectScoreUntouched() {
        Account a = TestDataFactory.accounts(1, false)[0];
        a.Score__c = 15;
        insert a;
        Datetime before = [SELECT LastModifiedDate FROM Account WHERE Id = :a.Id].LastModifiedDate;

        Test.startTest();
        Database.executeBatch(new AccountScoreBatch(), 200);
        Test.stopTest();

        Account after = [SELECT Score__c, LastModifiedDate FROM Account WHERE Id = :a.Id];
        Assert.areEqual(15, after.Score__c, 'Score unchanged');
        Assert.areEqual(before, after.LastModifiedDate, 'No DML when value already correct');
    }

    @IsTest
    static void batchCorrectsStaleScore() {
        Account a = TestDataFactory.accounts(1, false)[0];
        a.Score__c = 99;
        insert a;

        Test.startTest();
        Database.executeBatch(new AccountScoreBatch(), 200);
        Test.stopTest();

        Assert.areEqual(15, [SELECT Score__c FROM Account WHERE Id = :a.Id].Score__c, 'Stale score corrected');
    }

    @IsTest
    static void scoringHandlesNullsAndCaps() {
        Assert.areEqual(0, AccountScoring.score(new Account()), 'Nulls score zero');
        Assert.areEqual(100, AccountScoring.score(new Account(AnnualRevenue = 900000000, NumberOfEmployees = 100000)), 'Capped at 100');
    }

    @IsTest
    static void scheduleRegistersTheJob() {
        Test.startTest();
        String jobId = System.schedule('Score test', '0 0 2 * * ?', new AccountScoreSchedule());
        Test.stopTest();
        CronTrigger ct = [SELECT CronExpression FROM CronTrigger WHERE Id = :jobId];
        Assert.areEqual('0 0 2 * * ?', ct.CronExpression, 'Scheduled for 02:00');
    }
}
