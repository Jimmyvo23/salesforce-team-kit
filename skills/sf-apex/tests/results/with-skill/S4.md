**`force-app/main/default/classes/OpportunityService.cls`**
```apex
public with sharing class OpportunityService {
    private static final String PARTNER_RT_DEVELOPER_NAME = 'Partner';

    // Looked up by DeveloperName (no SOQL) because Ids differ between sandbox and production.
    public static Id partnerRecordTypeId {
        get {
            if (partnerRecordTypeId == null) {
                Schema.RecordTypeInfo info = Schema.SObjectType.Account
                    .getRecordTypeInfosByDeveloperName().get(PARTNER_RT_DEVELOPER_NAME);
                if (info == null) {
                    throw new IllegalArgumentException(
                        'Account record type with DeveloperName "' + PARTNER_RT_DEVELOPER_NAME + '" was not found.');
                }
                partnerRecordTypeId = info.getRecordTypeId();
            }
            return partnerRecordTypeId;
        }
        private set;
    }

    public static List<Opportunity> openForAccounts(Set<Id> accountIds) {
        return [SELECT Id, Name, StageName FROM Opportunity WHERE AccountId IN :accountIds AND IsClosed = false WITH USER_MODE];
    }

    public static List<Opportunity> openForPartnerAccounts(Set<Id> accountIds) {
        return [
            SELECT Id, Name, StageName
            FROM Opportunity
            WHERE AccountId IN :accountIds
            AND Account.RecordTypeId = :partnerRecordTypeId
            AND IsClosed = false
            WITH USER_MODE
        ];
    }
}
```

@IsTest
private class OpportunityServiceTest {
    @IsTest
    static void returnsOnlyOpenOpportunitiesOfPartnerAccounts() {
        Id partnerRtId = OpportunityService.partnerRecordTypeId;
        Account partner = new Account(Name = 'Partner Co', RecordTypeId = partnerRtId);
        Account other = new Account(Name = 'Other Co');
        insert new List<Account>{ partner, other };

        List<Opportunity> opps = new List<Opportunity>();
        for (Integer i = 0; i < 200; i++) {
            opps.add(new Opportunity(
                Name = 'P' + i, AccountId = partner.Id, StageName = 'Prospecting', CloseDate = Date.today().addDays(30)));
        }
        opps.add(new Opportunity(
            Name = 'Other', AccountId = other.Id, StageName = 'Prospecting', CloseDate = Date.today().addDays(30)));
        opps.add(new Opportunity(
            Name = 'Closed', AccountId = partner.Id, StageName = 'Closed Won', CloseDate = Date.today()));
        insert opps;

        Test.startTest();
        List<Opportunity> result = OpportunityService.openForPartnerAccounts(new Set<Id>{ partner.Id, other.Id });
        Test.stopTest();

        Assert.areEqual(200, result.size(), 'Only open opportunities on Partner accounts should be returned');
    }
}
