## `force-app/main/default/classes/AccountOpenCasesController.cls`
```apex
public with sharing class AccountOpenCasesController {
    @AuraEnabled(cacheable=true)
    public static List<Case> getOpenCases(Id accountId) {
        return [
            SELECT Id, CaseNumber, Subject, Status, Priority, CreatedDate
            FROM Case
            WHERE AccountId = :accountId AND IsClosed = false
            WITH USER_MODE
            ORDER BY CreatedDate DESC
            LIMIT 200
        ];
    }
}
```

private class AccountOpenCasesControllerTest {

    @IsTest
    static void returnsOnlyOpenCasesForTheAccount() {
        Account a = new Account(Name = 'Acme');
        Account other = new Account(Name = 'Other');
        insert new List<Account>{ a, other };
        insert new List<Case>{
            new Case(AccountId = a.Id, Subject = 'Open', Status = 'New'),
            new Case(AccountId = a.Id, Subject = 'Done', Status = closedStatus()),
            new Case(AccountId = other.Id, Subject = 'Elsewhere', Status = 'New')

        Test.startTest();
        List<Case> rows = AccountOpenCasesController.getOpenCases(a.Id);
        Test.stopTest();
        Assert.areEqual(1, rows.size(), 'Only the open case of this account');
        Assert.areEqual('Open', rows[0].Subject, 'Open case returned');
    }

    @IsTest
    static void returnsEmptyListWhenNoOpenCases() {
        Account a = new Account(Name = 'Acme');
        insert a;
        Test.startTest();
        List<Case> rows = AccountOpenCasesController.getOpenCases(a.Id);
        Test.stopTest();
        Assert.isTrue(rows.isEmpty(), 'No cases expected');
    }

    @IsTest
    static void handles200Cases() {
        Account a = new Account(Name = 'Acme');
        insert a;
        List<Case> cases = new List<Case>();
        for (Integer i = 0; i < 200; i++) {
            cases.add(new Case(AccountId = a.Id, Subject = 'Case ' + i, Status = 'New'));
        }
        insert cases;
        Test.startTest();
        List<Case> rows = AccountOpenCasesController.getOpenCases(a.Id);
        Test.stopTest();
        Assert.areEqual(200, rows.size(), 'All 200 open cases returned');
    }

        insert new Case(AccountId = a.Id, Subject = 'Open', Status = 'New', OwnerId = u.Id);
        List<Case> rows;

        System.runAs(u) {

        Test.stopTest();

## `…/lwc/accountOpenCases/accountOpenCases.js`
```js
import { LightningElement, api, wire } from 'lwc';
import getOpenCases from '@salesforce/apex/AccountOpenCasesController.getOpenCases';

const COLUMNS = [
    { label: 'Case', fieldName: 'url', type: 'url',
      typeAttributes: { label: { fieldName: 'CaseNumber' } } },
    { label: 'Subject', fieldName: 'Subject' },
    { label: 'Status', fieldName: 'Status' },
    { label: 'Priority', fieldName: 'Priority' }
];

export default class AccountOpenCases extends LightningElement {
    @api recordId;
    columns = COLUMNS;
    rows = [];
    error;
    loaded = false;

    @wire(getOpenCases, { accountId: '$recordId' })
    wiredCases({ data, error }) {
        if (data) {
            this.rows = data.map((c) => ({ ...c, url: `/lightning/r/Case/${c.Id}/view` }));
            this.error = undefined;
            this.loaded = true;
        } else if (error) {
            this.rows = [];
            this.error = error.body?.message ?? 'Unable to load cases';
            this.loaded = true;
        }
    }

    get isLoading() { return !this.loaded; }
    get isEmpty() { return this.loaded && !this.error && this.rows.length === 0; }
}
```

## `…/lwc/accountOpenCases/__tests__/accountOpenCases.test.js`

import { createElement } from '@lwc/engine-dom';
import AccountOpenCases from 'c/accountOpenCases';
import getOpenCases from '@salesforce/apex/AccountOpenCasesController.getOpenCases';

jest.mock(
    '@salesforce/apex/AccountOpenCasesController.getOpenCases',
    () => {
        const { createApexTestWireAdapter } = require('@salesforce/sfdx-lwc-jest');
        return { default: createApexTestWireAdapter(jest.fn()) };
    },
    { virtual: true }
);

const CASES = [
    { Id: '500000000000001AAA', CaseNumber: '00001001', Subject: 'Login', Status: 'New', Priority: 'High' },
    { Id: '500000000000002AAA', CaseNumber: '00001002', Subject: 'Billing', Status: 'Working', Priority: 'Low' }
];
const flushPromises = () => Promise.resolve();

describe('c-account-open-cases', () => {
    afterEach(() => {
        while (document.body.firstChild) document.body.removeChild(document.body.firstChild);
        jest.clearAllMocks();
    });

    function create() {
        const el = createElement('c-account-open-cases', { is: AccountOpenCases });
        el.recordId = '001000000000001AAA';
        document.body.appendChild(el);
        return el;
    }

    it('passes recordId to the wire and renders the rows', async () => {
        const el = create();
        getOpenCases.emit(CASES);
        await flushPromises();
        expect(getOpenCases.getLastConfig()).toEqual({ accountId: '001000000000001AAA' });
        const table = el.shadowRoot.querySelector('lightning-datatable');
        expect(table.data.length).toBe(2);
        expect(table.data[0].url).toBe('/lightning/r/Case/500000000000001AAA/view');
    });

    it('shows the empty state', async () => {
        const el = create();
        getOpenCases.emit([]);
        await flushPromises();
        expect(el.shadowRoot.querySelector('.empty')).not.toBeNull();
    });

    it('shows the error message', async () => {
        const el = create();
        getOpenCases.error({ message: 'No access' });
        await flushPromises();
        expect(el.shadowRoot.querySelector('.error').textContent).toBe('No access');
    });
});

## Notes
- **Security:** the controller is `with sharing` and its query uses `WITH USER_MODE`, so users only see Cases they have access to.
- **Access:** users need access to the Apex class through a permission set, plus read access to Case and its fields.
- **Cap:** the list is limited to 200 Cases.
- **Jest setup:** this project has no `package.json`. To run the Jest test, run `npm install --save-dev @salesforce/sfdx-lwc-jest` and add `"test:unit": "sfdx-lwc-jest"` to the scripts.
- **Apex test assumptions:** the tests assume the org has a `Standard User` profile and at least one closed Case status. They create their own data, because this project has no `TestDataFactory`.
- **Placement:** after deploying to your sandbox, add the component to the Account Lightning page in App Builder.
