# S5: Open cases LWC (with-skill)

I couldn't write any files because the Write tool is disabled in this session. All the code is below, with the path for each file. `sourceApiVersion` is 62.0, so I wrote the security settings explicitly.

The component calls a cacheable Apex method rather than using LDS. LDS's related-list wire can't cleanly filter on `IsClosed`, so Apex is the better fit here.

## `force-app/main/default/classes/AccountOpenCasesController.cls`
```apex
public with sharing class AccountOpenCasesController {
    @AuraEnabled(cacheable=true)
    public static List<Case> getOpenCases(Id accountId) {
        if (accountId == null) {
            return new List<Case>();
        }
        return [
            SELECT Id, CaseNumber, Subject, Status, Priority, CreatedDate
            FROM Case
            WHERE AccountId = :accountId AND IsClosed = false
            WITH USER_MODE
            ORDER BY CreatedDate DESC
            LIMIT 200
        ];
    }
}
```

## `force-app/main/default/classes/AccountOpenCasesControllerTest.cls`
```apex
@IsTest
private class AccountOpenCasesControllerTest {
    private static final Integer BULK = 200;

    @TestSetup
    static void setup() {
        // Closed status differs per org, so look up one whose IsClosed is true.
        String closedStatus = [SELECT MasterLabel FROM CaseStatus WHERE IsClosed = true LIMIT 1].MasterLabel;
        Account withCases = new Account(Name = 'With Cases');
        Account other = new Account(Name = 'Other');
        insert new List<Account>{ withCases, other };

        List<Case> cases = new List<Case>();
        for (Integer i = 0; i < BULK; i++) {
            cases.add(new Case(AccountId = withCases.Id, Subject = 'Open ' + i, Status = 'New'));
        }
        cases.add(new Case(AccountId = withCases.Id, Subject = 'Done', Status = closedStatus));
        cases.add(new Case(AccountId = other.Id, Subject = 'Other account', Status = 'New'));
        insert cases;
    }

    private static Account accountNamed(String name) {
        return [SELECT Id FROM Account WHERE Name = :name LIMIT 1];
    }

    @IsTest
    static void returnsOnlyOpenCasesForTheAccount() {
        Account a = accountNamed('With Cases');
        Test.startTest();
        List<Case> rows = AccountOpenCasesController.getOpenCases(a.Id);
        Test.stopTest();
        Assert.areEqual(BULK, rows.size(), 'All open cases, no closed and no other-account cases');
        for (Case c : rows) {
            Assert.areEqual('New', c.Status, 'Only open cases returned');
        }
    }

    @IsTest
    static void returnsEmptyWhenAccountHasNoOpenCases() {
        Account none = new Account(Name = 'No Cases');
        insert none;
        Test.startTest();
        List<Case> rows = AccountOpenCasesController.getOpenCases(none.Id);
        Test.stopTest();
        Assert.isTrue(rows.isEmpty(), 'No cases expected');
    }

    @IsTest
    static void returnsEmptyForNullAccountId() {
        Assert.isTrue(AccountOpenCasesController.getOpenCases(null).isEmpty(), 'Null id gives empty list');
    }

    @IsTest
    static void standardUserWithPermissionSetCanCall() {
        Account a = accountNamed('With Cases');
        Profile p = [SELECT Id FROM Profile WHERE Name = 'Standard User' LIMIT 1];
        User u = new User(
            Alias = 'nocase', Email = 'nocase@example.invalid',
            Username = 'nocase' + System.currentTimeMillis() + '@example.invalid',
            LastName = 'NoCase', ProfileId = p.Id, TimeZoneSidKey = 'UTC', LocaleSidKey = 'en_US',
            EmailEncodingKey = 'UTF-8', LanguageLocaleKey = 'en_US'
        );
        insert u;

        PermissionSet ps = new PermissionSet(Name = 'Open_Cases_Test', Label = 'Open Cases Test');
        insert ps;
        insert new SetupEntityAccess(
            ParentId = ps.Id,
            SetupEntityId = [SELECT Id FROM ApexClass WHERE Name = 'AccountOpenCasesController' LIMIT 1].Id
        );
        insert new PermissionSetAssignment(AssigneeId = u.Id, PermissionSetId = ps.Id);

        List<Case> rows;
        System.runAs(u) {
            Test.startTest();
            rows = AccountOpenCasesController.getOpenCases(a.Id);
            Test.stopTest();
        }
        Assert.isNotNull(rows, 'Call succeeds with the permission set');
        Assert.isTrue(rows.size() <= BULK, 'Sharing never widens the result');
    }
}
```

## `force-app/main/default/lwc/accountOpenCases/accountOpenCases.js`
```js
import { LightningElement, api, wire } from 'lwc';
import getOpenCases from '@salesforce/apex/AccountOpenCasesController.getOpenCases';

const COLUMNS = [
    { label: 'Case', fieldName: 'url', type: 'url',
      typeAttributes: { label: { fieldName: 'CaseNumber' } } },
    { label: 'Subject', fieldName: 'Subject' },
    { label: 'Status', fieldName: 'Status' },
    { label: 'Priority', fieldName: 'Priority' }
];

export default class AccountOpenCases extends LightningElement {
    @api recordId;
    columns = COLUMNS;
    rows = [];
    error;
    loaded = false;

    @wire(getOpenCases, { accountId: '$recordId' })
    wiredCases({ data, error }) {
        if (data) {
            this.rows = data.map((c) => ({ ...c, url: `/lightning/r/Case/${c.Id}/view` }));
            this.error = undefined;
            this.loaded = true;
        } else if (error) {
            this.rows = [];
            this.error = error.body?.message ?? 'Unable to load cases';
            this.loaded = true;
        }
    }

    get isLoading() {
        return !this.loaded;
    }

    get isEmpty() {
        return this.loaded && !this.error && this.rows.length === 0;
    }
}
```

## `…/accountOpenCases/__tests__/accountOpenCases.test.js`

```js
import { createElement } from '@lwc/engine-dom';
import AccountOpenCases from 'c/accountOpenCases';
import getOpenCases from '@salesforce/apex/AccountOpenCasesController.getOpenCases';

jest.mock(
    '@salesforce/apex/AccountOpenCasesController.getOpenCases',
    () => {
        const { createApexTestWireAdapter } = require('@salesforce/sfdx-lwc-jest');
        return { default: createApexTestWireAdapter(jest.fn()) };
    },
    { virtual: true }
);

const CASES = [
    { Id: '500000000000001AAA', CaseNumber: '00001001', Subject: 'Login', Status: 'New', Priority: 'High' },
    { Id: '500000000000002AAA', CaseNumber: '00001002', Subject: 'Billing', Status: 'Working', Priority: 'Low' }
];

const flushPromises = () => Promise.resolve();

describe('c-account-open-cases', () => {
    afterEach(() => {
        while (document.body.firstChild) document.body.removeChild(document.body.firstChild);
        jest.clearAllMocks();
    });

    function create() {
        const el = createElement('c-account-open-cases', { is: AccountOpenCases });
        el.recordId = '001000000000001AAA';
        document.body.appendChild(el);
        return el;
    }

    it('passes recordId to the wire and renders the rows', async () => {
        const el = create();
        getOpenCases.emit(CASES);
        await flushPromises();
        expect(getOpenCases.getLastConfig()).toEqual({ accountId: '001000000000001AAA' });
        const table = el.shadowRoot.querySelector('lightning-datatable');
        expect(table.data.length).toBe(2);
        expect(table.data[0].url).toBe('/lightning/r/Case/500000000000001AAA/view');
    });

    it('shows the empty state', async () => {
        const el = create();
        getOpenCases.emit([]);
        await flushPromises();
        expect(el.shadowRoot.querySelector('.empty')).not.toBeNull();
        expect(el.shadowRoot.querySelector('lightning-datatable')).toBeNull();
    });

    it('shows the error message', async () => {
        const el = create();
        getOpenCases.error({ message: 'No access' });
        await flushPromises();
        expect(el.shadowRoot.querySelector('.error').textContent).toBe('No access');
    });
});
```

## Notes
- **Access:** the query runs `WITH USER_MODE` in a `with sharing` class. Users need the Apex class granted through a permission set, plus read access to Case and the fields listed. If a user lacks field access, the query throws and the component shows the error message.
- **Row limit:** the list is capped at 200 rows. Tell me if you want paging.
- **Closed status:** the Apex test looks up a closed status from `CaseStatus` instead of hard-coding one.
- **Permission-set test:** the `runAs` test only shows that a Standard User with the permission set can call the method. It doesn't show that the access checks block anyone.
