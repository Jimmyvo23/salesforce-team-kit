```apex
@IsTest
private class OpportunityServiceTest {
    @TestSetup
    static void setup() {
        Account withOpps = new Account(Name = 'With Opps');
        Account other = new Account(Name = 'Other');
        insert new List<Account>{ withOpps, other };

        insert new List<Opportunity>{
            new Opportunity(Name = 'Open 1', AccountId = withOpps.Id, StageName = 'Prospecting', CloseDate = Date.today().addDays(30)),
            new Opportunity(Name = 'Open 2', AccountId = withOpps.Id, StageName = 'Qualification', CloseDate = Date.today().addDays(30)),
            new Opportunity(Name = 'Closed Won', AccountId = withOpps.Id, StageName = 'Closed Won', CloseDate = Date.today()),
            new Opportunity(Name = 'Other Open', AccountId = other.Id, StageName = 'Prospecting', CloseDate = Date.today().addDays(30))
        };
    }

    @IsTest
    static void openForAccountsReturnsOnlyOpenOpportunitiesForGivenAccounts() {
        Account withOpps = [SELECT Id FROM Account WHERE Name = 'With Opps'];

        Test.startTest();
        List<Opportunity> result = OpportunityService.openForAccounts(new Set<Id>{ withOpps.Id });
        Test.stopTest();

        Assert.areEqual(2, result.size(), 'Closed and other-account opportunities should be excluded');
        for (Opportunity opp : result) {
            Assert.areNotEqual('Closed Won', opp.StageName);
        }
    }

    @IsTest
    static void openForAccountsReturnsEmptyForEmptyInput() {
        Assert.isTrue(OpportunityService.openForAccounts(new Set<Id>()).isEmpty());
    }
}
```
