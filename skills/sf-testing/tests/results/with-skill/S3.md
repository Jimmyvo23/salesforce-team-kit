@IsTest
private class OpportunityServiceTest {
    @TestSetup
    static void setup() {
        Account a = new Account(Name = 'ZZ_TEST Main');
        Account b = new Account(Name = 'ZZ_TEST Other');
        insert new List<Account>{ a, b };
        insert new List<Opportunity>{
            new Opportunity(Name = 'ZZ_TEST open', AccountId = a.Id, StageName = 'Prospecting',
                CloseDate = Date.today().addDays(30), Partner_Discount__c = 12.50),
            new Opportunity(Name = 'ZZ_TEST closed', AccountId = a.Id, StageName = 'Closed Won',
                CloseDate = Date.today()),
            new Opportunity(Name = 'ZZ_TEST other', AccountId = b.Id, StageName = 'Prospecting',
                CloseDate = Date.today().addDays(30))
        };
    }

    static Account mainAccount() {
        return [SELECT Id FROM Account WHERE Name = 'ZZ_TEST Main'];
    }

    @IsTest
    static void returnsOnlyOpenOpportunities() {
        Id accId = mainAccount().Id;
        Test.startTest();
        List<Opportunity> result = OpportunityService.openForAccounts(new Set<Id>{ accId });
        Test.stopTest();
        Assert.areEqual(1, result.size(), 'Only the open Opportunity is returned');
        Assert.areEqual('ZZ_TEST open', result[0].Name, 'The closed Opportunity is excluded');
    }

    @IsTest
    static void partnerDiscountIsStoredWithTwoDecimals() {
        Test.startTest();
        Opportunity o = [SELECT Partner_Discount__c FROM Opportunity WHERE Name = 'ZZ_TEST open'];
        Test.stopTest();
        Assert.areEqual(12.50, o.Partner_Discount__c, 'Partner discount is saved as entered');
    }

    @IsTest
    static void emptyAccountSetReturnsNothing() {
        Test.startTest();
        List<Opportunity> result = OpportunityService.openForAccounts(new Set<Id>());
        Test.stopTest();
        Assert.isTrue(result.isEmpty(), 'No accounts, no Opportunities');
    }

    @IsTest
    static void otherAccountsOpportunitiesAreExcluded() {
        Id accId = mainAccount().Id;
        Test.startTest();
        List<Opportunity> result = OpportunityService.openForAccounts(new Set<Id>{ accId });
        Test.stopTest();
        for (Opportunity o : result) {
            Assert.areNotEqual('ZZ_TEST other', o.Name, 'Other Account Opportunity must not appear');
        }
    }

    @IsTest
    static void bulk200OpenOpportunitiesAreAllReturned() {
        Account bulk = new Account(Name = 'ZZ_TEST Bulk');
        insert bulk;
        List<Opportunity> opps = new List<Opportunity>();
        for (Integer i = 0; i < 200; i++) {
            opps.add(new Opportunity(Name = 'ZZ_TEST bulk ' + i, AccountId = bulk.Id,
                StageName = 'Prospecting', CloseDate = Date.today().addDays(30)));
        }
        insert opps;
        Test.startTest();
        List<Opportunity> result = OpportunityService.openForAccounts(new Set<Id>{ bulk.Id });
        Test.stopTest();
        Assert.areEqual(200, result.size(), 'All 200 open Opportunities are returned');
    }

    static User buildUser(Boolean withAccess) {
        Id profileId = [SELECT Id FROM Profile WHERE Name = 'Minimum Access - Salesforce' LIMIT 1].Id;
        String uniq = String.valueOf(Crypto.getRandomInteger());
        User u = new User(
            LastName = 'ZZ_TEST ' + uniq, Alias = 'zz' + uniq.left(5),
            Email = 'zztest' + uniq + '@example.com',
            Username = 'zztest' + uniq + '@example.com.test',
            ProfileId = profileId, TimeZoneSidKey = 'GMT', LocaleSidKey = 'en_US',
            EmailEncodingKey = 'UTF-8', LanguageLocaleKey = 'en_US');
        System.runAs(new User(Id = UserInfo.getUserId())) {
            insert u;
            if (withAccess) {
                PermissionSet ps = new PermissionSet(Label = 'ZZ_TEST Opp Read', Name = 'ZZ_TEST_Opp_Read' + uniq);
                insert ps;
                insert new List<SObject>{
                    new ObjectPermissions(ParentId = ps.Id, SobjectType = 'Opportunity',
                        PermissionsRead = true, PermissionsViewAllRecords = true),
                    new ObjectPermissions(ParentId = ps.Id, SobjectType = 'Account',
                        PermissionsRead = true, PermissionsViewAllRecords = true),
                    new FieldPermissions(ParentId = ps.Id, SobjectType = 'Opportunity',
                        Field = 'Opportunity.Partner_Discount__c', PermissionsRead = true, PermissionsEdit = false)
                };
                insert new PermissionSetAssignment(AssigneeId = u.Id, PermissionSetId = ps.Id);
            }
        }
        return u;
    }

    @IsTest
    static void userWithReadAccessSeesOpenOpportunities() {
        User u = buildUser(true);
        Id accId = mainAccount().Id;
        List<Opportunity> result;
        Test.startTest();
        System.runAs(u) {
            result = OpportunityService.openForAccounts(new Set<Id>{ accId });
        }
        Test.stopTest();
        Assert.areEqual(1, result.size(), 'User with read access sees the open Opportunity');
    }

    @IsTest
    static void userWithoutOpportunityAccessGetsQueryException() {
        User u = buildUser(false);
        Id accId = mainAccount().Id;
        Boolean thrown = false;
        Test.startTest();
        System.runAs(u) {
            Assert.isFalse(Schema.SObjectType.Opportunity.isAccessible(), 'Precondition: no Opportunity access');
            try {
                OpportunityService.openForAccounts(new Set<Id>{ accId });
            } catch (QueryException e) {
                thrown = true;
            }
        }
        Test.stopTest();
        Assert.isTrue(thrown, 'USER_MODE blocks a user without Opportunity read');
    }
}
