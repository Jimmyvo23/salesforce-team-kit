```apex
@IsTest
private class OpportunityServiceTest {
    @TestSetup
    static void setup() {
        insert new List<Account>{ TestDataFactory.account('Target'), TestDataFactory.account('Other') };
        Id target = [SELECT Id FROM Account WHERE Name = 'Target' LIMIT 1].Id;
        Id other = [SELECT Id FROM Account WHERE Name = 'Other' LIMIT 1].Id;
        List<Opportunity> opps = new List<Opportunity>();
        opps.addAll(TestDataFactory.opportunities(target, 2, false, null));
        opps.addAll(TestDataFactory.opportunities(target, 1, true, null));
        opps.addAll(TestDataFactory.opportunities(other, 1, false, null));
        insert opps;
    }

    static Id accountId(String name) {
        return [SELECT Id FROM Account WHERE Name = :name LIMIT 1].Id;
    }

    @IsTest
    static void partnerDiscountSavesPercentValue() {
        Opportunity o = TestDataFactory.opportunities(accountId('Other'), 1, false, 12.5)[0];
        Test.startTest();
        insert o;
        Test.stopTest();
        Assert.areEqual(12.5, [SELECT Partner_Discount__c FROM Opportunity WHERE Id = :o.Id].Partner_Discount__c,
            'Partner discount is stored as entered');
    }

    @IsTest
    static void partnerDiscountStaysBlankWhenNotSet() {
        Opportunity o = TestDataFactory.opportunities(accountId('Other'), 1, false, null)[0];
        Test.startTest();
        insert o;
        Test.stopTest();
        Assert.isNull([SELECT Partner_Discount__c FROM Opportunity WHERE Id = :o.Id].Partner_Discount__c,
            'No default is written');
    }

    @IsTest
    static void bulkInsertOf200OppsKeepsPartnerDiscount() {
        List<Opportunity> rows = TestDataFactory.opportunities(accountId('Other'), 200, false, 7.25);
        Test.startTest();
        insert rows;
        Test.stopTest();
        Assert.areEqual(200, [SELECT COUNT() FROM Opportunity WHERE Partner_Discount__c = 7.25],
            'All 200 keep the discount');
    }

    @IsTest
    static void openForAccountsReturnsOnlyOpenOppsOfThoseAccounts() {
        Id target = accountId('Target');
        Test.startTest();
        List<Opportunity> result = OpportunityService.openForAccounts(new Set<Id>{ target });
        Test.stopTest();
        Assert.areEqual(2, result.size(), 'Only the 2 open Opps of Target');
        for (Opportunity o : result) {
            Assert.areNotEqual('Closed Won', o.StageName, 'No closed Opps');
        }
    }

    @IsTest
    static void openForAccountsWithNoIdsReturnsEmpty() {
        Test.startTest();
        List<Opportunity> result = OpportunityService.openForAccounts(new Set<Id>());
        Test.stopTest();
        Assert.isTrue(result.isEmpty(), 'Empty input, empty result');
    }

    // ASSUMES: Minimum Access - Salesforce has no Opportunity access, so WITH USER_MODE throws.
    @IsTest
    static void userWithoutOpportunityAccessCannotQuery() {
        Profile p = [SELECT Id FROM Profile WHERE Name = 'Minimum Access - Salesforce' LIMIT 1];
        String unique = String.valueOf(Crypto.getRandomLong()).remove('-');
        User u = new User(Alias = 'noopp', LastName = 'NoAccess', Email = 'noopp' + unique + '@example.com',
            Username = 'noopp' + unique + '@example.com.test', ProfileId = p.Id, EmailEncodingKey = 'UTF-8',
            LanguageLocaleKey = 'en_US', LocaleSidKey = 'en_US', TimeZoneSidKey = 'America/Los_Angeles');
        System.runAs(new User(Id = UserInfo.getUserId())) { insert u; }
        Id target = accountId('Target');
        System.runAs(u) {
            Assert.isFalse(Schema.SObjectType.Opportunity.isAccessible(), 'Precondition: no Opportunity access');
            Test.startTest();
            try {
                OpportunityService.openForAccounts(new Set<Id>{ target });
                Assert.fail('Expected QueryException');
            } catch (QueryException e) {
                Assert.isTrue(e.getMessage().length() > 0, 'User mode blocks the query');
            }
            Test.stopTest();
        }
    }

    // ASSUMES: the story wants Partner_Discount__c read-only for ordinary users.
    @IsTest
    static void userWithReadOnlyPartnerDiscountCannotEditIt() {
        User u = RestrictedUser.create('Opportunity',
            new List<String>{ 'Name', 'StageName', 'CloseDate', 'AccountId' },
            new List<String>{ 'Partner_Discount__c' });
        Account acct = TestDataFactory.account('Owned by restricted');
        acct.OwnerId = u.Id;
        insert acct;
        Opportunity o = TestDataFactory.opportunities(acct.Id, 1, false, null)[0];
        System.runAs(u) {
            Assert.isFalse(Schema.SObjectType.Opportunity.fields.Partner_Discount__c.isUpdateable(),
                'Precondition: the user cannot edit the field');
            insert as user o;
            Test.startTest();
            try {
                update as user new Opportunity(Id = o.Id, Partner_Discount__c = 50);
                Assert.fail('Expected DmlException');
            } catch (DmlException e) {
                Assert.isTrue(e.getMessage().length() > 0, 'FLS blocks the edit');
            }
            Test.stopTest();
        }
        Assert.isNull([SELECT Partner_Discount__c FROM Opportunity WHERE Id = :o.Id].Partner_Discount__c,
            'Value unchanged');
    }
}
```
